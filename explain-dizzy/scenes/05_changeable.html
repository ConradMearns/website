pub fn execute_transfer(context: &execute_transfer_context, command: TransferFunds) -> Result<(), Error> {
    let balance = (context.query.get_balance)(GetBalanceInput { account: command.payer.clone() })?;
    if balance.amount < command.amount {
        (context.emit.transfer_rejected)(TransferRejected { payer: command.payer, amount: command.amount });
    } else {
        (context.emit.funds_debited)(FundsDebited { payer: command.payer, amount: command.amount });
        (context.emit.funds_credited)(FundsCredited { payee: command.payee, amount: command.amount });
    }
    Ok(())
}
