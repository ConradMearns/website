with import <nixpkgs> { };

let
  tex = (pkgs.texlive.combine {
    inherit (pkgs.texlive) scheme-full
      markdown
      xcolor
      geometry xstring
      blindtext
      ;
  });
in
stdenv.mkDerivation {
  name = "node";
  buildInputs = [
    tex
    ...
  ];
...
}
